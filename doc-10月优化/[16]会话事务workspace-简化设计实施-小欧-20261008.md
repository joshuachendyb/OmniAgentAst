```python
# -*- coding: utf-8 -*-
"""会话 Workspace 存储与落值（[16] 独立模块，不进 storage.py）"""

def _normalize_ws_path(p: str) -> str:
    """归一化唯一源：空串/非绝对路径硬拒 → resolve 绝对化 + 去尾分隔符 + Windows casefold。

    v1.3 P1-4（运行时实测新发现，比 v1.1 P1-2 严重）：`Path("").expanduser().resolve()`
    **不抛异常且返回进程 cwd** —— 实测返回 `F:\\OmniAgentAs-repair\\backend`。若不拦，
    `PATCH [""]` 会把**整个后端启动目录**静默入库授权（当前 cwd 恰好在代码库根内会被
    `_is_forbidden_path` 拦成 non_system；但 cwd 一旦不在禁区——如从项目根启动——
    即等于"授权整个启动目录"，且用户以为传了个空串。故显式硬拒。
    ⚠ 仍可能抛 OSError/ValueError（**实测**断开的 UNC 抛 `OSError [WinError 64]`；
    含空字节/400字超长/`<>|` 非法字符实测均不抛）——调用方必须捕获，失败归入
    rejected 走 422，不得冒泡成 500（v1.1 P1-2）。"""
    if not p or not p.strip():
        raise ValueError("路径为空")
    _raw = p.strip()
    if not Path(_raw).is_absolute():
        raise ValueError(f"须为绝对路径(相对路径会随进程 cwd 漂移): {_raw}")
    return str(Path(_raw).expanduser().resolve()).rstrip("\\/").casefold()


def get_session_workspaces(conn, session_id: str) -> list:
    """读该会话全部授权目录行。ORDER BY 保证默认行在前，rowid 即插入序。"""
    return conn.execute(
        "SELECT path, is_default FROM chat_session_workspaces "
        "WHERE session_id = ? ORDER BY is_default DESC, rowid",
        (session_id,),
    ).fetchall()


def _ws_precheck(paths: list[str]) -> list[dict]:
    """禁区预检：复用 path_safe_check._is_forbidden_path()（全项目唯一判据）。
    返回被拒清单，不抛异常（调用方按决策 3 整单 422）。"""
    from app.tools.security.path_safe_check import _is_forbidden_path  # 延迟导入避环
    rejected = []
    for p in paths:
        category, msg = _is_forbidden_path(p)
        if category:
            rejected.append({"path": p, "reason": msg})
    return rejected


def _ws_apply(conn, session_id: str, paths: list[str]) -> None:
    """三步式覆盖写（决策 5）——**PATCH 覆盖写的唯一写库实现**（v1.2 校准：懒建幂等占行走
    `INSERT OR IGNORE`，见决策 4，不经此处——本函数含 DELETE，非原子，不可用于并发占行）。
    SELECT 旧行 → DELETE 全删 → executemany INSERT；is_default 回查保留；不碰 updated_at。"""
    old = {r["path"]: r["is_default"] for r in conn.execute(
        "SELECT path, is_default FROM chat_session_workspaces WHERE session_id=?",
        (session_id,))}
    conn.execute("DELETE FROM chat_session_workspaces WHERE session_id=?", (session_id,))
    if not paths:
        return
    conn.executemany(
        "INSERT INTO chat_session_workspaces(session_id, path, is_default) VALUES (?, ?, ?)",
        [(session_id, p, old.get(p, 0)) for p in paths])


def set_session_workspaces(conn, session_id: str, raw_paths: list[str]) -> list[dict]:
    """**PATCH 唯一写入口**（决策 1/2）：归一化 → 去重 → 预检 → 写库。
    返回被拒清单；非空即调用方返 422（决策 3），空即 200。
    v1.1 P1-2：归一化单条失败不 500——捕获 OSError/ValueError 归入 rejected 一并报出
    （v1.3 P1-4：空串/相对路径亦由此归入 rejected，不单独开校验分支）。
    v1.3 P1-6：**含 `[]` 清空在内**的任何成功写，都把 `ws_initialized` 置 1——
    这是"用户显式清空"与"从未配置过"的分界，缺它则决策 2「[]=清空」被决策 4 懒建当场复活
    （清空后下一条消息必重新 mkdir + 插回 is_default=1，且每次多一个目录）。"""
    paths, rejected = [], []
    for p in raw_paths:
        try:
            paths.append(_normalize_ws_path(p))
        except (OSError, ValueError) as _e:
            rejected.append({"path": p, "reason": f"路径非法: {type(_e).__name__}: {_e}"})
    if rejected:
        return rejected                      # 归一化失败即整单 422，不再写库
    paths = list(dict.fromkeys(paths))       # 归一化后去重（a\b 与 a/b 归一为同一项）
    rejected = _ws_precheck(paths)
    if rejected:
        return rejected
    _ws_apply(conn, session_id, paths)
    conn.execute("UPDATE chat_sessions SET ws_initialized = 1 WHERE id = ?", (session_id,))
    return []


def _load_workspaces_conn(conn, session_id: str) -> list[tuple[str, int]]:
    """任务快照源（决策 9）：读行 → 归一化 → 双重过滤（exists + 禁区兜底）。
    conn 由调用方经 db.atxn 传入，本函数内禁自建连接。空会话/无行 → []。

    v1.3 P1-3：本函数是**每次任务启动**都会跑的读路径，`_normalize_ws_path` 在此**零防护**
    （v1.1 P1-2 只护了 PATCH 写侧）→ 存量行里一条断开的 UNC 就能抛 OSError。故此处
    与写侧同口径捕获（(OSError, ValueError)）：单行坏行跳过 + 告警，不让一行脏数据
    掀掉整个任务启动（P0-1 落点错位时更不能被它带着一起炸）。
    v1.3 P1-7：`exists()` 落空的行**必须留告警日志** —— 否则目录被改名/移动/E 盘没挂时，
    徽标仍显示 N 而运行时实际生效 0（GET 说 N、agent 用 0），且因行还在懒建也不会补建，
    永久错位到用户手动开面板改为止。"""
    from app.tools.security.path_safe_check import _is_forbidden_path  # 延迟导入避环
    out = []
    for r in get_session_workspaces(conn, session_id):
        try:
            p = _normalize_ws_path(r["path"])
        except (OSError, ValueError) as _e:
            logger.warning(f"[workspace] 存量行归一化失败(跳过该行): {r['path']!r} ({type(_e).__name__}: {_e})")
            continue
        if not Path(p).exists():
            logger.warning(f"[workspace] 授权目录不存在(跳过, 但列表仍显示): {p}")
            continue
        if _is_forbidden_path(p)[0]:
            logger.warning(f"[workspace] 存量授权目录已落入禁区，本任务不放行: {p}")
            continue
        out.append((p, r["is_default"]))
    return out


def load_effective_workspaces(conn, session_id: str) -> tuple[list, list]:
    """**GET 唯一读源**（决策 1）：返回 `(有效集, 不可用清单)`，两者与任务启动快照**同口径**。
    v1.3 P2-①：GET 过去直接返裸行，而运行时 `_load_workspaces_conn` 还要再过
    `exists()` + `_is_forbidden_path` 两道闸 → "徽标显示 N、agent 实际生效 M" 的静默错位。
    本函数复用同一个 `_load_workspaces_conn` 判据（DRY：过滤逻辑只有一份），
    把"库里有但本轮不生效"的行如实回给前端，由面板提示用户改，而不是让用户对着
    一个生效不了的目录排查一整天。"""
    from app.tools.security.path_safe_check import _is_forbidden_path
    effective, unusable = [], []
    for r in get_session_workspaces(conn, session_id):
        try:
            p = _normalize_ws_path(r["path"])
        except (OSError, ValueError):
            unusable.append({"path": r["path"], "reason": "路径非法"})
            continue
        if not Path(p).exists():
            unusable.append({"path": p, "reason": "目录不存在"})
            continue
        if _is_forbidden_path(p)[0]:
            unusable.append({"path": p, "reason": "已落入禁区"})
            continue
        effective.append((p, r["is_default"]))
    return effective, unusable


def resolve_ws_initialized(conn, session_id: str) -> bool:
    """会话是否已配置过 workspace（v1.3 P1-6 决策 4 的前置判据）。
    口径仿 `storage.resolve_session_interject`（北京老陈 [11] 3.2 立的"会话级唯一真源"惯例）：
    存量行 ALTER 后为 NULL → COALESCE 读作 0（未配置），故懒建照跑，不影响老会话首启。"""
    row = conn.execute(
        "SELECT COALESCE(ws_initialized, 0) FROM chat_sessions WHERE id = ?", (session_id,)
    ).fetchone()
    return bool(row[0]) if row else False


def session_exists(conn, session_id: str) -> bool:
    """会话存在性（v1.1 P1-1）：存在且未删才放行，否则调用方返 404。
    必要性：PRAGMA foreign_keys=ON 下，对不存在 session_id 的 INSERT 会抛
    sqlite3.IntegrityError → 500。口径与 session_service 一致（is_deleted = FALSE）。"""
    row = conn.execute(
        "SELECT 1 FROM chat_sessions WHERE id = ? AND is_deleted = FALSE",
        (session_id,),
    ).fetchone()
    return row is not None
```
