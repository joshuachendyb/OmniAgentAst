### 3.2 `app/services/chat/workspace_service.py` — **新建模块**（全部 7 个函数）

**为什么独立模块**：[15] 把 7 个函数塞进 986 行 45 函数的 `storage.py` 是 SRP 违规（自认；10-09 实测已 1037 行 48 函数，债在加重）。本模块自给自足，只依赖 `db` / `logger` / `time_utils` / `path_safe_check._is_forbidden_path`（延迟导入）。

```python
# -*- coding: utf-8 -*-
"""会话 Workspace 存储与落值（[16] 独立模块，不进 storage.py）"""
from pathlib import Path
import sqlite3
from typing import Optional

from app.logger import logger
from app.db import db
from app.utils.time_utils import get_local_iso_timestamp

WORKSPACE_REJECTED_FIELD = "workspace_rejected"


def _normalize_ws_path(p: str) -> str:
    """归一化唯一源：resolve 绝对化 + 去尾分隔符 + Windows casefold。
    ⚠ 可能抛 OSError/ValueError（非法字符/空字节/超长路径）——调用方必须捕获，
    失败归入 rejected 走 422，不得冒泡成 500（v1.1 P1-2）。"""
    return str(Path(p).expanduser().resolve()).rstrip("\\/").casefold()


def get_session_workspaces(conn, session_id: str) -> list:
    """读该会话全部授权目录行。ORDER BY 保证默认行在前，rowid 即插入序。"""
    return conn.execute(
        "SELECT path, is_default FROM chat_session_workspaces "
        "WHERE session_id = ? ORDER BY is_default DESC, rowid",
        (session_id,),
    ).fetchall()


def _ws_precheck(paths: list[str]) -> list[dict]:
    """禁区预检：复用 path_safe_check._is_forbidden_path()（全项目唯一判据）。
    返回被拒清单，不抛异常（调用方按决策 3 整单 422）。"""
    from app.tools.security.path_safe_check import _is_forbidden_path  # 延迟导入避环
    rejected = []
    for p in paths:
        category, msg = _is_forbidden_path(p)
        if category:
            rejected.append({"path": p, "reason": msg})
    return rejected


def _ws_apply(conn, session_id: str, paths: list[str]) -> None:
    """三步式覆盖写（决策 5）——唯一写库实现。
    SELECT 旧行 → DELETE 全删 → executemany INSERT；is_default 回查保留；不碰 updated_at。"""
    old = {r["path"]: r["is_default"] for r in conn.execute(
        "SELECT path, is_default FROM chat_session_workspaces WHERE session_id=?",
        (session_id,))}
    conn.execute("DELETE FROM chat_session_workspaces WHERE session_id=?", (session_id,))
    if not paths:
        return
    conn.executemany(
        "INSERT INTO chat_session_workspaces(session_id, path, is_default) VALUES (?, ?, ?)",
        [(session_id, p, old.get(p, 0)) for p in paths])


def set_session_workspaces(conn, session_id: str, raw_paths: list[str]) -> list[dict]:
    """**PATCH 唯一写入口**（决策 1/2）：归一化 → 去重 → 预检 → 写库。
    返回被拒清单；非空即调用方返 422（决策 3），空即 200。
    v1.1 P1-2：归一化单条失败不 500——捕获 OSError/ValueError 归入 rejected 一并报出。"""
    paths, rejected = [], []
    for p in raw_paths:
        try:
            paths.append(_normalize_ws_path(p))
        except (OSError, ValueError) as _e:
            rejected.append({"path": p, "reason": f"路径非法: {type(_e).__name__}: {_e}"})
    if rejected:
        return rejected                      # 归一化失败即整单 422，不再写库
    paths = list(dict.fromkeys(paths))       # 归一化后去重（a\b 与 a/b 归一为同一项）
    rejected = _ws_precheck(paths)
    if rejected:
        return rejected
    _ws_apply(conn, session_id, paths)
    return []


def _load_workspaces_conn(conn, session_id: str) -> list[tuple[str, int]]:
    """任务快照源（决策 9）：读行 → 归一化 → 双重过滤（exists + 禁区兜底）。
    conn 由调用方经 db.atxn 传入，本函数内禁自建连接。空会话/无行 → []。"""
    from app.tools.security.path_safe_check import _is_forbidden_path  # 延迟导入避环
    out = []
    for r in get_session_workspaces(conn, session_id):
        p = _normalize_ws_path(r["path"])
        if not Path(p).exists():
            continue
        if _is_forbidden_path(p)[0]:
            logger.warning(f"[workspace] 存量授权目录已落入禁区，本任务不放行: {p}")
            continue
        out.append((p, r["is_default"]))
    return out


def session_exists(conn, session_id: str) -> bool:
    """会话存在性（v1.1 P1-1）：存在且未删才放行，否则调用方返 404。
    必要性：PRAGMA foreign_keys=ON 下，对不存在 session_id 的 INSERT 会抛
    sqlite3.IntegrityError → 500。口径与 session_service 一致（is_deleted = FALSE）。"""
    row = conn.execute(
        "SELECT 1 FROM chat_sessions WHERE id = ? AND is_deleted = FALSE",
        (session_id,),
    ).fetchone()
    return row is not None
```
