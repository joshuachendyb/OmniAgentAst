## 一、研究范围与方法

1. **研究范围**：`backend/app/api/v1/` 全部路由文件、`backend/app/monitoring/`（监控独立库）、`backend/app/services/task/`（任务注册/状态）、`backend/app/main.py` 路由挂载、`frontend/src/` 现有页面与菜单。
2. **研究方法**：逐文件实读路由定义（`@router.get/post/...`）与响应模型字段，核对前端 `services/api/` 已接通的接口，找出"后端有数据、前端未展示"的信息面。
3. **研究问题**：
   - 后端现有哪些信息面可展示？
   - 哪些已有 API 但前端未用（零后端开发即可上屏）？
   - 哪些信息有价值但缺 API（需少量后端新增）？

## 二、研究结论（先说重点）

1. 后端现有 **6 类可直接展示的信息面**，全部有现成 API；
2. 其中 3 类（运行指标、token 全局用量、任务中心数据）**前端尚未展示**，属"临门一脚"；
3. 任务中心数据**完全现成**——`GET /tasks` 全局列表端点已存在（v1.0 误判"缺列表端点"，v1.1 已核实更正：task_queries.py:37，`tasks` 表自带 status/total_operations/success_count/failed_count 字段），任务监控页**后端零新增**；
4. 首选落地方案：**侧边菜单单一"任务监控"页（四区布局：关键指标 / 运行指标 / Token 用量 / 任务中心，后端零改动）**。顶栏服务状态已由 `components/Layout/ServiceStatusPill` 实现（不在本报告设计范围），①区健康卡直接复用其 AppContext 快照，不另建健康灯、不另发 /health 轮询。

来源：`app/monitoring/agent_telemetry.py`（帧类型口径在 `M_SKIP` 集合中定义）与 `agent_runner` 的发布链路。

## 五、有价值但目前未暴露 API 的信息（需小量后端新增）

| 信息 | 现状 | 需要的工作 |

## 六、前端展示方案：任务监控菜单页（顶栏已实现，本报告只设计菜单页）

### 6.1 顶栏状态展示（已实现，不在本设计范围）

顶栏服务状态已由 `frontend/src/components/Layout/ServiceStatusPill` 落地：胶囊三态（绿正常 / 黄缓慢 / 红离线），悬浮浮层给地址、响应耗时、应答时刻、数据库状态、进程信息；数据源为 `AppContext` 的 `healthStatus` / `refreshHealth`（挂载自检 + 点击重检，非定时轮询）。本报告只设计侧边菜单「任务监控」页；①区健康卡复用同一份 AppContext 快照，不新增顶栏元素、不另建 /health 轮询。

### 6.2 侧边菜单新增：单一「任务监控」菜单页

**v1.1 修订**：北京老陈定案——不在侧边菜单拆"仪表盘"和"任务中心"两个菜单，**合并为一个「任务监控」菜单**，两类内容放同一页面，UI 分块布局。

- 菜单项位置：插入**现有左侧边栏菜单**（对话任务 / 文件管理 / 知识库(置灰) / 历史会话 / 快捷指令 / 设置页面），建议置于「设置页面」上方、第二个 divider 之后，路由 `/monitor`；
- 页面形态（方案C，对齐 Settings2 最新先例）：`frontend/src/pages/TaskMonitor/index.tsx` 只做**路由薄壳**（页标题 + 四区组装 + 取数 + 区块错误态），业务组件与取数 hook 落 `frontend/src/features/monitor/{components,hooks}`；
- **后端零新增**（v1.1 核实：任务列表端点已存在）。

```ts
// 编辑历史:
// 2026-11-02 小欧 v1.8 - 新建: 监控页取数层 useMonitorPolling(10s 静默轮询, JSON diff 防闪屏, retry)
// 2026-11-03 小欧 v1.9 - 删 useHealthPolling/healthText: 顶栏状态已由 ServiceStatusPill+AppContext 落地,
//   健康卡直接读 AppContext 快照, 页面不再持有第二份 /health 轮询 — 小欧-2026-11-03
import { useEffect, useRef, useState } from 'react';

/** 区块取数三态: data=最新数据(失败时保留旧值), error=最近一次请求失败, retry=立即重拉 */
export interface PollState<T> {
  data: T | null;
  error: boolean;
  retry: () => void;
}

/** 通用 10s 静默轮询 + JSON diff(值未变不 setState) — 小欧 2026-11-02
 *  retry() 给区块级「重试」按钮用: 换 tick 触发 effect 重跑(重设 interval 一并归零)。 */
export function useMonitorPolling<T>(fetcher: () => Promise<T>): PollState<T> {
  const [data, setData] = useState<T | null>(null);
  const [error, setError] = useState(false);
  const [tick, setTick] = useState(0);
  const fetcherRef = useRef(fetcher); // 恒最新取数器, effect 不依赖其身份
  fetcherRef.current = fetcher;

  useEffect(() => {
    let alive = true;
    const run = async () => {
      try {
        const next = await fetcherRef.current();
        if (!alive) return;
        setData((prev) => (JSON.stringify(prev) === JSON.stringify(next) ? prev : next));
        setError(false);
      } catch {
        // _silent 封装默认无 toast, 区块级 Alert 呈现
        if (alive) setError(true);
      }
    };
    void run();
    const intervalId = setInterval(() => void run(), 10_000);
    return () => {
      alive = false;
      clearInterval(intervalId);
    };
  }, [tick]);

  return { data, error, retry: () => setTick((n) => n + 1) };
}
```

import React from 'react';
import { Card, Col, Empty, Progress, Row } from 'antd';
import type { MetricsSummaryResponse } from '@/services/api/monitor.api';
import { Colors, FontSize, Radius, Spacing } from '@/utils/stepStyles';

/** 秒 -> ms 展示(直方图单位为秒: http_request_duration_seconds) */
const toMs = (sec: number | undefined): string =>
  sec === undefined ? '--' : `${(sec * 1000).toFixed(1)}ms`;

const cardBody = { padding: Spacing.XL } as const;
const cardTitle = (text: string) => <span style={{ fontSize: FontSize.SECONDARY }}>{text}</span>;

const HttpMetricsPanel: React.FC<{ metrics: MetricsSummaryResponse }> = ({ metrics }) => {
  // 近1小时滚动窗口(collector retention=3600); 无流量严禁渲染假100% — 小欧 2026-11-02 v1.7
  const total = metrics.metrics.http_requests_total?.count ?? 0;
  const errors = metrics.metrics.errors_total?.count ?? 0;
  const hasTraffic = total > 0;
  const successRate = hasTraffic ? Math.max(0, Math.round((1 - errors / total) * 1000) / 10) : 0;
  // 比例条是图形元素, 用功能图形色(≥90 绿 / ≥70 橙 / 其余红); 文字色规则不适用于描边
  const rateColor =
    successRate >= 90 ? Colors.SUCCESS : successRate >= 70 ? Colors.WAIT_ACTION : Colors.ERROR;
  const dur = metrics.metrics.http_request_duration_seconds;
  const percentileRows: Array<[string, string]> = [
    ['avg', toMs(dur?.avg)],
    ['p50', toMs(dur?.p50)],
    ['p90', toMs(dur?.p90)],
    ['p95', toMs(dur?.p95)],
    ['p99', toMs(dur?.p99)],
  ];
  return (
    <Row gutter={[16, 16]}>
      <Col xs={24} md={10}>
        <Card
          style={{ height: '100%', borderRadius: Radius.LG }}
          bodyStyle={cardBody}
          title={cardTitle('请求成功率（近 1 小时）')}
        >
          {hasTraffic ? (
            <Progress
              percent={successRate}
              strokeColor={rateColor}
              format={(p) => `成功率 ${p}%`}
            />
          ) : (
            <Empty description="近 1 小时无请求" />
          )}
        </Card>
      </Col>
      <Col xs={24} md={14}>
        <Card
          style={{ height: '100%', borderRadius: Radius.LG }}
          bodyStyle={cardBody}
          title={cardTitle('请求延迟分位（近 1 小时）')}
        >
          {percentileRows.map(([label, val], i) => (
            <div
              key={label}
              style={{
                display: 'flex',
                justifyContent: 'space-between',
                fontSize: FontSize.PRIMARY,
                padding: `${Spacing.XS}px 0`,
                borderBottom:
                  i === percentileRows.length - 1 ? 'none' : `1px solid ${Colors.BORDER.LIGHT}`,
              }}
            >
              <span style={{ color: Colors.TEXT.PRIMARY }}>{label}</span>
              <span style={{ color: Colors.TEXT.STRONG, fontVariantNumeric: 'tabular-nums' }}>
                {val}
              </span>
            </div>
          ))}
        </Card>
      </Col>
    </Row>
  );
};

export default HttpMetricsPanel;

const STATUS_META = {
  pending: { color: 'default', text: '待执行' },
  executing: { color: 'processing', text: '进行中' },
  success: { color: 'success', text: '已完成' },
  failed: { color: 'error', text: '失败' },
  rolled_back: { color: 'warning', text: '已回滚' },

type TaskStatusKey = keyof typeof STATUS_META;

const statusTag = (status: string): React.ReactNode => {

};

  text,
  first,

    orientation="left"
    style={{

/** 区块三态: 失败→Alert(重试) | 首帧→Spin | 就绪→内容 */
const Zone: React.FC<{ name: string; state: PollState<unknown>; children: React.ReactNode }> = ({
  name,
  state,
  children,
}) =>
  state.error ? (
    <Alert
      type="error"
      showIcon
      message={`${name}获取失败`}
      action={
        <Button size="small" onClick={state.retry}>
          重试
        </Button>
      }
    />
  ) : state.data === null ? (
    <div style={centerSpin}>
      <Spin />
    </div>
  ) : (
    <>{children}</>
  );

const TaskMonitorPage: React.FC = () => {
  // ① 健康卡: 吃 AppContext 快照(与顶栏胶囊同一份, 全应用只有这一处 /health 状态);
  //   页面挂载时刷一次保证不是旧快照, 其余时间不轮询 — 小欧 2026-11-03 v1.9
  const { healthStatus, refreshHealth } = useApp();
  useEffect(() => {
    void refreshHealth();
  }, [refreshHealth]);

  const metrics = useMonitorPolling(() => monitorApi.getMetricsSummary());
  const token = useMonitorPolling(() => monitorApi.getGlobalTokenUsage());

  // ④ 任务列表: 打开页面首拉 + 手动刷新(不轮询); 同一份 state 同时喂①区"近期任务"卡

  const [tasksError, setTasksError] = useState(false);
  const [tasksLoading, setTasksLoading] = useState(false);

  const refreshTasks = useCallback(async () => {
    setTasksLoading(true);
    try {
      setTasks(await monitorApi.getRecentTasks(50));
      setTasksError(false);
    } catch (e) {
      setTasksError(true);
      handleError(e, { message: '任务列表获取失败' });
    } finally {
      setTasksLoading(false);
    }
  }, []);

  useEffect(() => {
    void refreshTasks();
  }, [refreshTasks]);

  const tasksState: PollState<TaskListRow[]> = {
    data: tasks,
    error: tasksError,
    retry: refreshTasks,
  };

  return (
    <div style={{ maxWidth: 1440, margin: '0 auto', padding: Spacing.XL }}>
      <Typography.Title level={3} style={{ marginTop: 0 }}>
        <MonitorOutlined /> 任务监控
      </Typography.Title>

      <ZoneTitle first icon={<AppstoreOutlined />} text="关键指标" />

      <ZoneTitle icon={<DashboardOutlined />} text="运行指标" />
      <Zone name="运行指标" state={metrics}>
        {metrics.data && <HttpMetricsPanel metrics={metrics.data} />}
      </Zone>

      <ZoneTitle icon={<FundOutlined />} text="Token 用量" />
      <Zone name="Token 用量" state={token}>
        {token.data && <TokenPanel usage={token.data} />}
      </Zone>

      <ZoneTitle icon={<UnorderedListOutlined />} text="任务中心" />
      <TaskCenterTable tasks={tasks ?? []} loading={tasksLoading} onRefresh={refreshTasks} />
    </div>
  );
};

export default TaskMonitorPage;
