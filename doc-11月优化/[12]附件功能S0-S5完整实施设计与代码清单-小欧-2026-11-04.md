### 3.6.1 v2 修订：`save_upload` 拆出 `save_bytes`（S4 前置）

```python
def _extract_richdoc(stored: Path, suffix: str) -> str:
    """调工具层读取器并取出纯数据段; 工具返回 error 时抛出。"""
    resp = _call_tool_reader(stored, suffix)
    status = ((resp.get("llm_data") or {}).get("status") or {})
    if status.get("exec_code") == "error":
        raise RuntimeError(status.get("detail") or "抽取失败")
    data = resp.get("data") or {}
    if suffix == ".xlsx":
        return _sheet_text(data)
    text = str(data.get("text") or "").strip()
    if not text:
        raise RuntimeError("未抽取到文本（可能是扫描件或空文档）")
    return text


def _call_tool_reader(stored: Path, suffix: str) -> Dict[str, Any]:
    """延迟导入: 三个读取器各自带 pdfplumber/docx/openpyxl, 顶层导入会拖慢服务启动。"""
    if suffix == ".pdf":
        from app.tools.document.read_pdf import read_pdf
        return read_pdf(str(stored))
    if suffix == ".docx":
        from app.tools.document.read_docx import read_docx
        return read_docx(str(stored))
    from app.tools.document.read_xlsx import read_xlsx
    return read_xlsx(str(stored))


def _sheet_text(data: Dict[str, Any]) -> str:
    """工作簿 → 文本表: 每 sheet 一段, 单 sheet 最多 RICH_MAX_ROWS 行。"""
    sheets = data.get("sheets") or ([data] if "rows" in data else [])
    blocks: List[str] = []
    for sh in sheets:
        name = sh.get("sheet_name") or "Sheet"
        headers = ["" if h is None else str(h) for h in (sh.get("headers") or [])]
        all_rows = sh.get("rows") or []
        rows = all_rows[:RICH_MAX_ROWS]
        lines = [" | ".join(headers)] if headers else []
        lines += [" | ".join("" if v is None else str(v) for v in row)
                  for row in rows]
        tail = "" if len(all_rows) <= RICH_MAX_ROWS else \
            f"\n…（仅注入前 {RICH_MAX_ROWS} 行，完整数据请用文件工具读取）"
        blocks.append(f"## {name}（{len(all_rows)} 行）\n" + "\n".join(lines) + tail)
    if not blocks:
        raise RuntimeError("工作簿为空")
    return "\n\n".join(blocks)
```

+            (new_name, pack_id),

+            fail(PACK_NAME_CONFLICT)
+        conn.execute(

+             cur["group_name"] if group_name is None else (group_name or "").strip(),
+             get_local_iso_timestamp(), pack_id),
+        )
+        if items is not None:

+            _insert_items(conn, pack_id, _clean_items(items))

+        ).fetchone():

+        conn.execute("DELETE FROM material_packs WHERE id = ?", (pack_id,))

```diff
+      if (enqueue(rec)) result.ok += 1;
+      else overLimit += 1;
+    } catch {
+      // 后端已按 code 给出具体原因（类型/大小），但用户此刻只需要知道"这个文件没进去"
+      message.error(`「${f.name}」上传失败，请重试`);
+      result.failed += 1;
+    }
+  }
+  if (overLimit > 0) message.warning(refLimitText());
+  result.failed += overLimit;
+  return result;

+/**
+ * 从剪贴板 / 拖拽的 DataTransfer 取文件（S3）。
+ *
+ * 粘贴截图时浏览器给的 File.name 为空串，直接上传会落成无名文件；
+ * 这里补一个带时间戳的名字（扩展名取 mime 子类型），落盘与 chip 都可读。
+ * 粘贴纯文本时 dt.files 为空 → 返回 [] → 调用方不 preventDefault，粘贴文字行为零变化。
+ */
+export function filesFromDataTransfer(dt: DataTransfer | null): File[] {
+  if (!dt) return [];
+  const stamp = new Date()
+    .toISOString()
+    .replace(/[-:T]/g, '')
+    .slice(0, 14);
+  const out: File[] = [];
+  let unnamed = 0;
+  for (const f of Array.from(dt.files)) {
+    if (f.name) {
+      out.push(f);
+      continue;
+    }
+    unnamed += 1;
+    const sub = (f.type.split('/')[1] || 'png').replace('jpeg', 'jpg');
+    out.push(
+      new File([f], `粘贴截图-${stamp}-${unnamed}.${sub}`, {
+        type: f.type || 'image/png',
+      })
+    );
+  }
+  return out;
+}

+ * ③「存为附件」（S4）——概览产出物行的动作实现。
+ *
+ * 两种后端结果都要落成**一条普通引用**，用户视角只有一个结果：「它进引用区了」。
+ * referenced：文件已在会话工作区 → 走 quoteFile（复用它的大小/存在性探测与提示），不复制文件。
+ * copied    ：文件在工作区外 → 后端已复制并登记，这里直接把返回记录入队。
+ */
+export async function collectArtifact(
+  sessionId: string,
+  sourcePath: string
+): Promise<boolean> {
+  try {
+    const r = await attachmentApi.collect(sessionId, sourcePath);
+    if (r.mode === 'referenced') {
+      const q = await quoteFile(r.path, { label: r.label, size: r.size });
+      if (!q.ok) {
+        message.warning('该产物已在工作区，但引用失败，请稍后重试');
+        return false;
+      }
+      message.success(`已引用「${r.label}」到对话`);
+      return true;
+    }
+    const pushed = enqueue(r.attachment);
+    if (!pushed) {
+      message.warning(refLimitText());
+      return false;
+    }
+    message.success(`已存为附件「${r.attachment.original_name}」`);
+    return true;
+  } catch {
+    message.error('存为附件失败：产物可能已被移动或删除');
+    return false;
+  }
+}
```

**要点**

1. `uploadAttachments` **不抛异常**：三个入口的宿主各不相同（antd 内部 / React 事件 / 面板动作），统一"内部提示 + 返回计数"比让每个调用方各写 try/catch 简单。
2. `filesFromDataTransfer` 只处理 `dt.files`，不读 `dt.items`：截图与文件拖拽都会落到 `files`；`items` 还要处理 `getAsFile()` 的异步回调，属多余复杂度（YAGNI）。
3. 批量失败**逐条提示**，不汇总成一句"3 个失败"——用户要据此决定重试哪一个。
4. 超 `MAX_REFS` 单独计数并只提示一次，避免 10 条文件刷 10 条相同警告。

```tsx
  /** 只接管"带文件"的拖拽; 拖文本、拖链接一律放行, 不打扰浏览器默认行为 */
  const hasFiles = (dt: DataTransfer | null): boolean =>
    !!dt && Array.from(dt.types).includes('Files');

  const onDragEnter = (e: React.DragEvent): void => {
    if (!hasFiles(e.dataTransfer)) return;
    dragDepth.current += 1;
    setDragOver(true);
  };

  const onDragLeave = (): void => {
    dragDepth.current = Math.max(0, dragDepth.current - 1);
    if (dragDepth.current === 0) setDragOver(false);
  };

  const onDragOver = (e: React.DragEvent): void => {
    if (!hasFiles(e.dataTransfer)) return;
    e.preventDefault(); // 不阻止 → 松手时浏览器直接打开该文件, 当前页面被替换
    e.dataTransfer.dropEffect = 'copy';
  };

  const onDrop = (e: React.DragEvent): void => {
    if (!hasFiles(e.dataTransfer)) return;
    e.preventDefault();
    dragDepth.current = 0;
    setDragOver(false);
    handleFiles(filesFromDataTransfer(e.dataTransfer));
  };

  /** 粘贴截图: 只有剪贴板里真有文件才接管; 纯文本粘贴不 preventDefault(否则正文粘不进来) */
  const onPaste = (e: React.ClipboardEvent): void => {
    const files = filesFromDataTransfer(e.clipboardData);
    if (files.length === 0) return;
    e.preventDefault();
    handleFiles(files);
  };
```

**要点**

1. `hasFiles` 用 `dataTransfer.types` 判，不用 `dt.files.length`：拖拽进入阶段 `files` 在某些浏览器为空，只有 `types` 可靠；松手后两者一致。
2. `onDragOver` 必须 `preventDefault()`，否则 `drop` 事件根本不会触发（HTML5 规范），页面还会跳去打开文件。
3. 粘贴必须"有文件才 preventDefault"：无条件阻止会把普通文本粘贴也吞掉（输入框粘字失效），这是粘贴截图最容易踩的坑。
4. 三个入口共用 `handleFiles` → 配额、D1 建会话、失败提示只有一份（DRY），S3 只需新增事件接线，不碰上传逻辑。
