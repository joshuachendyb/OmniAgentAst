124|def compute_severity_score(level_counts: Counter, total: int) -> float:
125|    """严重度评分: 0-100, 越高越严重"""
126|    if total == 0:
127|        return 0.0
128|    weighted_sum = sum(SEVERITY_WEIGHTS.get(lvl, 0) * cnt for lvl, cnt in level_counts.items())
129|    max_possible = total * SEVERITY_WEIGHTS["ERROR"]
130|    score = (weighted_sum / max_possible) * 100 if max_possible > 0 else 0.0
131|    return round(score, 2)

142|def detect_anomalies(entries: list[LogEntry], hourly: dict[str, int]) -> tuple[list[str], list[str]]:
143|    """识别异常峰值时段和高频错误时段"""
144|    if not hourly:
145|        return [], []
146|
147|    counts = list(hourly.values())
148|    avg_count = sum(counts) / len(counts) if counts else 0
149|    std_dev = (sum((c - avg_count) ** 2 for c in counts) / len(counts)) ** 0.5 if len(counts) > 1 else 0
150|
151|    anomaly_hours = []
152|    high_error_hours = []
153|
154|    error_by_hour: dict[str, int] = defaultdict(int)
155|    for e in entries:
156|        if e.level == "ERROR":
157|            error_by_hour[e.hour_key] += 1
158|
159|    for hour_str, count in hourly.items():
160|        # 异常峰值: 超过平均值+1倍标准差
161|        if std_dev > 0 and count > avg_count + std_dev:
162|            anomaly_hours.append(hour_str)
163|        # 高频错误: 该小时ERROR数 >= 3
164|        if error_by_hour.get(hour_str, 0) >= 3:
165|            high_error_hours.append(hour_str)
166|
167|    return sorted(anomaly_hours), sorted(high_error_hours)

170|def aggregate_patterns(entries: list[LogEntry], top_n: int = 5) -> tuple[list[dict], list[dict]]:
171|    """聚合相同错误/警告消息, 找出TOP N"""
172|    error_msgs: Counter = Counter()
173|    warning_msgs: Counter = Counter()
174|
175|    for e in entries:
176|        norm_msg = e.normalized_message
177|        if e.level == "ERROR":
178|            error_msgs[norm_msg] += 1
179|        elif e.level == "WARNING":
180|            warning_msgs[norm_msg] += 1
181|
182|    def build_top(counter: Counter) -> list[dict]:
183|        result = []
184|        for msg, count in counter.most_common(top_n):
185|            # 根因分析
186|            root_cause = infer_root_cause(msg)
187|            result.append({
188|                "message": msg,
189|                "count": count,
190|                "percentage": round(count / len([e for e in entries if e.level in ("ERROR", "WARNING")]) * 100, 2) if any(e.level in ("ERROR", "WARNING") for e in entries) else 0,
191|                "root_cause": root_cause,
192|            })
193|        return result
194|
195|    return build_top(error_msgs), build_top(warning_msgs)
