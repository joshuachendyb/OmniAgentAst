import re
from collections import Counter

def count_errors(log_path):
    with open(log_path, 'r', encoding='utf-8') as f:
        lines = f.readlines()
    errors = [l for l in lines if re.search(r'ERROR|FATAL|Exception', l, re.I)]
    return len(errors), Counter(errors)
