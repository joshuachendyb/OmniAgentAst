## 一、问题背景：为什么要做这个优化

这几天调试前后端配合的功能，总是后端到处修改，而且还总是错误百出。北京老陈追问：

1. 与前端配合的部分零散分布，到处都放，应该集中
2. 支持前端配合的代码逻辑层次/架构层次是错误的
3. 后端yield信息给前端，为什么需要那么多代码来处理这个事情？

**核心问题**：一定是后端与前端配合的这些逻辑的烂代码导致的。

## 三、病根分析：架构层次问题

### 3.1 后端在做前端的事

**当前架构（错误）**：

## 四、优化逻辑：怎么把分散的yield集中起来

### 4.1 核心思路：分层分离

**当前（混乱）**：handler直接yield前端Step

```
action_handler.py
├── 业务：创建确认 → confirm_id
├── 前端：构造MetaStep(type="paused")    ← 混在一起！
└── 前端：yield给前端显示               ← 混在一起！
```

**优化后（清晰）**：handler只返回业务结果，统一层转换

```
action_handler.py（只做业务）
├── 业务：创建确认 → confirm_id
└── 返回：{"status":"paused", "confirm_id": xxx}

        ↓

event_emitter.py（统一转换）
├── 读取业务结果
├── 根据status构造前端Step
└── yield给前端显示
```

# answer_handler.py 优化后（只做业务，不yield Step）
async def handle_answer(agent, parsed: dict) -> dict:
    """处理LLM返回的答案/错误/未知类型
    返回：{"action": "completed/failed/retrying", ...}
    不yield任何Step，由event_emitter统一转换
    """
    if parsed is None:
        parsed = {}
    
    step = agent.llm_call_count
    parsed_type = parsed.get("type", "answer")
    
    # 错误类型处理
    if parsed_type == "error":
        content, _, err_type = _parse_answer(parsed)
        agent._consecutive_reasoning_only = 0
        agent.message_builder.add_assistant_message(content)
        set_failed(agent, content)
        
        # 只返回业务结果
        return {
            "action": "failed",
            "response": "任务执行失败",
            "error_type": err_type,
            "error_message": content
        }
    
    # 未知类型处理
    if parsed_type != "answer":
        agent._consecutive_reasoning_only = 0
        content = parsed.get("content", "") or parsed.get("thought", "") or ""
        if content:
            agent.message_builder.add_assistant_message(f"[无效响应:{parsed_type}] {content}")
        set_failed(agent, f"LLM返回未知响应类型: {parsed_type}")
        
        return {
            "action": "failed",
            "response": f"LLM返回未知响应类型: {parsed_type}",
            "error_type": "unknown_response",
            "error_message": content
        }
    
    # 正常回答处理
    content = format_tool_call_markup(parsed.get("content", "") or "")
    reasoning = format_tool_call_markup(parsed.get("reasoning", "") or "")
    
    # 业务逻辑：空内容检测
    if not content and not reasoning:
        agent._consecutive_reasoning_only = 0
        return {"action": "retrying", "wait_time": 1}
    
    # 业务逻辑：重复检测
    if len(content) >= REPEAT_CHECK_MIN_LEN:
        deduped = _dedup_repeat(content)
        if deduped != content:
            content = deduped
    
    # 业务逻辑：推理空转防御
    if not content and reasoning:
        agent._consecutive_reasoning_only += 1
        if agent._consecutive_reasoning_only > REASONING_ONLY_MAX_ROUNDS:
            set_failed(agent, "连续推理轮次超限")
            return {
                "action": "failed",
                "response": "连续推理轮次超限",
                "error_type": "reasoning_only_limit",
                "error_message": f"连续{REASONING_ONLY_MAX_ROUNDS}轮仅返回推理内容"
            }
        # 正常推理，返回thought
        return {"action": "thought", "content": content, "reasoning": reasoning}
    
    # 正常回答
    agent._consecutive_reasoning_only = 0
    agent.message_builder.add_assistant_message(content)
    set_completed(agent)
    
    return {
        "action": "completed",
        "response": content,
        "reasoning": reasoning
    }
```

# sandbox_gate.py 优化后（只做业务，不yield Step）
async def sandbox_resolve(agent, step, call, tool_name, params, pre, safety_result, denied_list) -> dict:
    """沙箱预检结果处置
    返回：{"action": "passthrough/blocked/user_confirm/confirmed/rejected", ...}
    不yield任何Step，由event_emitter统一转换
    """
    # 预检通过
    if pre and pre.safe:
        return {"action": "passthrough"}
    
    # 危险拦截
    if pre and pre.blocked:
        denied_list.append((tool_name, f"沙箱预检未通过: {pre.blocked_reason}", call))
        return {
            "action": "blocked",
            "reason": f"沙箱预检未通过: {pre.blocked_reason}",
            "tool_name": tool_name
        }
    
    # 用户裁决
    if pre and pre.needs_user_confirm:
        confirm_id = await create_confirmation(agent.task_id, tool_name)
        
        # 业务逻辑：超时配置计算
        from app.config import get_config as _get_cfg
        _backend_timeout = int(float(_get_cfg().get("security.hitl_timeout", HITL_TIMEOUT)))
        _confirm_timeout = max(HITL_MIN_CONFIRM_TIMEOUT, _backend_timeout - HITL_CONFIRM_LEAD)
        
        # 返回业务结果，等待用户确认
        return {
            "action": "user_confirm",
            "confirm_id": confirm_id,
            "tool_name": tool_name,
            "params": {k: v for k, v in params.items() if k not in _SENSITIVE_FIELDS},
            "safety_level": "destructive",
            "trust_path": extract_trust_path(tool_name, params),
            "confirm_timeout": _confirm_timeout,
            "backend_timeout": _backend_timeout
        }
    
    # 异常兜底：直通
    return {"action": "passthrough"}


async def sandbox_resolve_after_confirm(agent, step, confirm_id, auth_result, denied_list) -> dict:
    """用户确认后的处理（从sandbox_resolve拆出）
    返回：{"action": "confirmed/rejected", ...}
    """
    if auth_result.get("confirmed"):
        return {"action": "confirmed", "confirm_id": confirm_id}
    else:

            "reason": "用户拒绝沙箱执行",

# react_cycle.py 优化后（纯调度，不yield前端Step）
async def _dispatch_handler(agent, llm_response) -> dict:
    """按type分派handler，获取业务结果
    返回：{"action": "...", ...} 业务结果
    不yield任何Step，由上层调用event_emitter统一转换
    """
    parsed_type = llm_response.get("type", "answer")
    
    # 根据类型分派handler，获取业务结果
    if parsed_type == "action":
        result = await handle_action(agent, llm_response)
    else:
        result = await handle_answer(agent, llm_response)
    
    # 返回业务结果，不yield
    return result


async def run_react_cycle(agent, task, context=None, max_steps=None, task_id=None, start_time=None):
    """ReAct循环：调用LLM→解析→分派handler→获取业务结果→转换Step"""
    from app.services.agent.event_emitter import emit_from_business_result
    
    # 初始化运行状态
    chunk_buffer = initialize_run_state(agent, task, task_id, context)
    
    while agent.llm_call_count < max_steps:
        # 取消/暂停检测
        if task_id:
            if await check_cancelled(task_id):
                break
            await wait_for_resume(task_id)
        
        # 单步处理：调用LLM获取响应
        llm_response = await call_llm(agent, chunk_buffer)
        
        # 分派handler，获取业务结果
        result = await _dispatch_handler(agent, llm_response)
        
        # 统一转换：业务结果 → 前端Step
        for s in emit_from_business_result(agent, agent.llm_call_count, result):
            yield s  # yield给SSE层转发
        
        # 根据业务结果更新状态
        action = result.get("action")
        if action == "completed":
            set_completed(agent)
            break
        elif action == "failed":
            set_failed(agent, result.get("error_message", ""))
            break
        elif action == "retrying":
            set_status(agent, AgentStatus.RETRYING, "触发重试")
            agent._retry_count = getattr(agent, '_retry_count', 0) + 1
            if agent._retry_count > 3:
                set_failed(agent, "重试超限")
                break
        elif action == "thought":
            # 推理内容，继续下一轮
            pass
        
        # chunk缓冲区超时检查
        if chunk_buffer.should_force_stop():
            set_failed(agent, f"chunk累积超时({agent.llm_call_count}步)")
            break
    
    # 循环结束处理
    if agent.status not in (AgentStatus.COMPLETED, AgentStatus.FAILED, AgentStatus.CANCELLED):
        set_cancelled(agent)
```

### 5.4 event_emitter.py 新建（统一转换层）

```python
# event_emitter.py 新建
"""统一转换层：业务结果 → 前端Step"""

from typing import Generator
from app.services.agent.steps.base import MetaStep, FinalStep, ThoughtStep, ThoughtStartStep


def emit_from_business_result(agent, step: int, result: dict) -> Generator:
    """统一转换：业务结果 → 前端Step
    
    所有前端展示逻辑集中在此，handler只返回业务结果
    """
    action = result.get("action")
    
    # --- answer_handler 返回的结果 ---
    if action == "completed":
        # 正常回答完成
        yield agent._step_emitter.emit(ThoughtStartStep(step=step))
        for _s in agent._step_emitter.emit_final_with_stats(FinalStep(
            step=step,
            response=result.get("response", ""),
            outcome="completed",
            reasoning=result.get("reasoning", ""),
        )):
            yield _s
    
    elif action == "failed":
        # 任务失败
        yield agent._step_emitter.emit(ThoughtStartStep(step=step))
        for _s in agent._step_emitter.emit_final_with_stats(FinalStep(
            step=step,
            response=result.get("response", "任务执行失败"),
            outcome="failed",
            error_type=result.get("error_type", ""),
            error_message=result.get("error_message", ""),
        )):
            yield _s
    
    elif action == "retrying":
        # 系统重试
        yield agent._step_emitter.emit(MetaStep(
            step=step,
            type="retrying",
            content=f"系统重试中...",
            wait_time=result.get("wait_time", 1),
            severity="info"
        ))
    
    elif action == "thought":
        # 推理内容
        yield agent._step_emitter.emit(ThoughtStartStep(step=step))
        yield agent._step_emitter.emit(ThoughtStep(
            step=step,
            content=result.get("content", ""),
            reasoning=result.get("reasoning", ""),
        ))
    
    # --- sandbox_gate 返回的结果 ---
    elif action == "passthrough":
        # 沙箱直通，不需要yield
        pass
    
    elif action == "blocked":
        # 安全拦截
        yield agent._step_emitter.emit(MetaStep(
            step=step,
            type="blocked",
            content=result.get("reason", "安全策略拦截"),
            tool_name=result.get("tool_name", ""),
            severity="warning"
        ))
    
    elif action == "user_confirm":
        # 需要用户确认
        yield agent._step_emitter.emit(MetaStep(
            step=step,
            type="paused",
            content=f"需要用户确认工具执行: {result.get('tool_name', '')}",
            confirm_id=result.get("confirm_id"),
            tool_name=result.get("tool_name"),
            params=result.get("params"),
            safety_level=result.get("safety_level"),
            trust_path=result.get("trust_path"),
            confirm_timeout=result.get("confirm_timeout"),
            backend_timeout=result.get("backend_timeout"),
            severity="attention"
        ))
    
    elif action == "confirmed":
        # 用户确认后恢复
        yield agent._step_emitter.emit(MetaStep(
            step=step,
            type="resumed",
            content="用户确认执行",
            confirm_id=result.get("confirm_id"),
            severity="info"
        ))
    
    elif action == "rejected":
        # 用户拒绝
        yield agent._step_emitter.emit(MetaStep(
            step=step,
            type="error",
            content=result.get("reason", "用户拒绝执行"),
            error_type="user_rejected",
            tool_name=result.get("tool_name", ""),
            severity="warning"
        ))
    
    # --- action_handler 返回的结果 ---
    elif action == "executing":
        # 工具执行中（由action_handler内部处理，此处不yield）
        pass
    
    # --- 未知action ---
    else:
        yield agent._step_emitter.emit(MetaStep(
            step=step,
            type="error",
            content=f"未知业务结果: {action}",
            error_type="unknown_action",
            severity="warning"
        ))
```
